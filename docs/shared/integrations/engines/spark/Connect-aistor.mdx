Open source Spark reaches AIStor Tables through the Iceberg REST catalog with SigV4 signing. Spark talks to two endpoints on the same host, so the keys appear twice: the `rest.*` keys sign catalog requests and the `s3.*` keys read the files.

```bash title="spark-sql against AIStor Tables"
spark-sql \
  --packages org.apache.iceberg:iceberg-spark-runtime-3.5_2.12:1.10.0,org.apache.iceberg:iceberg-aws-bundle:1.10.0 \
  --conf spark.sql.extensions=org.apache.iceberg.spark.extensions.IcebergSparkSessionExtensions \
  --conf spark.sql.catalog.aistor=org.apache.iceberg.spark.SparkCatalog \
  --conf spark.sql.catalog.aistor.type=rest \
  --conf spark.sql.catalog.aistor.uri=http://<aistor-host>:9000/_iceberg \
  --conf spark.sql.catalog.aistor.warehouse=olake \
  --conf spark.sql.catalog.aistor.rest.sigv4-enabled=true \
  --conf spark.sql.catalog.aistor.rest.signing-name=s3tables \
  --conf spark.sql.catalog.aistor.rest.signing-region=local \
  --conf spark.sql.catalog.aistor.rest.access-key-id=<access-key> \
  --conf spark.sql.catalog.aistor.rest.secret-access-key=<secret-key> \
  --conf spark.sql.catalog.aistor.io-impl=org.apache.iceberg.aws.s3.S3FileIO \
  --conf spark.sql.catalog.aistor.s3.endpoint=http://<aistor-host>:9000 \
  --conf spark.sql.catalog.aistor.s3.path-style-access=true \
  --conf spark.sql.catalog.aistor.s3.access-key-id=<access-key> \
  --conf spark.sql.catalog.aistor.s3.secret-access-key=<secret-key> \
  --conf spark.sql.catalog.aistor.client.region=us-east-1
```
