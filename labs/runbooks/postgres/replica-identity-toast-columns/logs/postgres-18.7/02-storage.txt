## The two rows of documents: four columns built from md5 output, and summary, 100,000 characters of one letter
SELECT id, length(body) AS body_chars, pg_column_size(body) AS body_stored,
       length(attachment) AS attachment_bytes, pg_column_size(attachment) AS attachment_stored,
       pg_column_size(meta) AS meta_stored, pg_column_size(tags) AS tags_stored,

## PostgreSQL 17 and later: is each value stored out of line?
SELECT id, pg_column_toast_chunk_id(body) IS NOT NULL AS body_out_of_line, pg_column_toast_chunk_id(attachment) IS NOT NULL AS attachment_out_of_line,
       pg_column_toast_chunk_id(meta) IS NOT NULL AS meta_out_of_line, pg_column_toast_chunk_id(tags) IS NOT NULL AS tags_out_of_line,
       pg_column_toast_chunk_id(summary) IS NOT NULL AS summary_out_of_line
