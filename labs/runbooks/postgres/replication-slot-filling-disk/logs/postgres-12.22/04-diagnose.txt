## Which slot holds the WAL, and is anything connected to it?
SELECT slot_name, slot_type, database, active, active_pid, restart_lsn, confirmed_flush_lsn,
       pg_size_pretty(pg_wal_lsn_diff(pg_current_wal_lsn(), restart_lsn)) AS retained_wal
FROM pg_replication_slots ORDER BY slot_name;
