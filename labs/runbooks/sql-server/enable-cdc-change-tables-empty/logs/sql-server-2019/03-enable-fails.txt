
## 02-fail-not-sysadmin.sql
CREATE DATABASE lab_perm;
CREATE LOGIN lab_owner_login WITH PASSWORD = N'Lab_Passw0rd_2026', CHECK_POLICY = OFF;

CREATE TABLE dbo.t (id INT NOT NULL PRIMARY KEY);
CREATE USER lab_owner_user FOR LOGIN lab_owner_login;
ALTER ROLE db_owner ADD MEMBER lab_owner_user;
CREATE USER lab_reader_user FOR LOGIN lab_reader_login;
ALTER ROLE db_datareader ADD MEMBER lab_reader_user;

-- A member of db_owner who is not sysadmin tries to enable the database.
EXECUTE AS LOGIN = N'lab_owner_login';

## 04-fail-ddl-trigger.sql
CREATE DATABASE lab_trigger;
CREATE TABLE dbo.lab_ddl_audit (logged_at DATETIME2(0) NOT NULL DEFAULT SYSUTCDATETIME(), who SYSNAME NOT NULL);

CREATE TRIGGER lab_audit_ddl ON DATABASE FOR DDL_DATABASE_LEVEL_EVENTS AS
BEGIN
  INSERT INTO master.dbo.lab_ddl_audit (who) VALUES (USER_NAME());
END;

EXEC sys.sp_cdc_enable_db;
Msg 22830, Level 16, State 1, Server mssql, Procedure sys.sp_cdc_enable_db_internal, Line 258
Could not update the metadata that indicates database lab_trigger is enabled for Change Data Capture. The failure occurred when executing the command 'create user cdc'. The error returned was 229: 'The INSERT permission was denied on the object 'lab_ddl_audit', database 'master', schema 'dbo'.'. Use the action and error to determine the cause of the failure and resubmit the request.

SELECT name, is_cdc_enabled FROM sys.databases WHERE name = N'lab_trigger';
name|is_cdc_enabled
----|--------------
lab_trigger|0

(1 rows affected)

-- Fix: disable the trigger, enable CDC, enable the trigger again.
DISABLE TRIGGER lab_audit_ddl ON DATABASE;

EXEC sys.sp_cdc_enable_db;

ENABLE TRIGGER lab_audit_ddl ON DATABASE;

SELECT name, is_cdc_enabled FROM sys.databases WHERE name = N'lab_trigger';
name|is_cdc_enabled
----|--------------
lab_trigger|1

(1 rows affected)

DISABLE TRIGGER lab_audit_ddl ON DATABASE;

EXEC sys.sp_cdc_disable_db;

DROP DATABASE lab_trigger;
DROP TABLE dbo.lab_ddl_audit;
