-- The WAL volume is written to a table that no task captures, so the tools have nothing to apply.
CREATE TABLE filler (id bigserial PRIMARY KEY, customer_id int NOT NULL, status text NOT NULL, note text) WITH (autovacuum_enabled = off);
CREATE PROCEDURE write_wal(target_mb int) LANGUAGE plpgsql AS $$
DECLARE
  start_lsn pg_lsn := pg_current_wal_lsn();
BEGIN
  WHILE pg_wal_lsn_diff(pg_current_wal_lsn(), start_lsn) < target_mb::numeric * 1024 * 1024 LOOP
    INSERT INTO filler (customer_id, status, note)
    SELECT g % 1000, 'new', repeat(md5(g::text), 8) FROM generate_series(1, 5000) g;
    COMMIT;
    IF (SELECT count(*) FROM filler) >= 50000 THEN
      TRUNCATE filler;
      COMMIT;
    END IF;
  END LOOP;
END $$;
